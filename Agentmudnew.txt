import asyncio
from typing import Dict, List, Optional
from redis import asyncio as aioredis
from app.models.conversation import *

class ConversationEngine:
    def __init__(self):
        self.redis: Optional[aioredis.Redis] = None
        self.channels: Dict[ChannelType, ConversationChannel] = {}
        self.websocket_connections: Dict[str, List] = {}
        
    async def initialize(self):
        self.redis = await aioredis.from_url("redis://localhost:6379", decode_responses=True)
        await self.initialize_channels()
        
    async def initialize_channels(self):
        for channel_type in ChannelType:
            self.channels[channel_type] = ConversationChannel(channel_type=channel_type)
            
    async def add_message(self, message: Message) -> bool:
        """Add a message to a channel and broadcast to subscribers"""
        channel = self.channels.get(message.channel)
        if not channel:
            return False
            
        channel.messages.append(message)
        
        # Persist to Redis for durability
        await self.redis.rpush(
            f"channel:{message.channel}:messages",
            message.model_dump_json()
        )
        
        # Trim message history
        if len(channel.messages) > 1000:
            channel.messages = channel.messages[-1000:]
            
        # Broadcast to WebSocket subscribers
        await self.broadcast_message(message)
        return True
        
    async def broadcast_message(self, message: Message):
        """Broadcast message to all WebSocket subscribers"""
        if message.channel in self.websocket_connections:
            for websocket in self.websocket_connections[message.channel]:
                try:
                    await websocket.send_text(message.model_dump_json())
                except:
                    # Remove broken connections
                    self.websocket_connections[message.channel].remove(websocket)
                    
    async def get_channel_history(self, channel: ChannelType, limit: int = 100) -> List[Message]:
        """Get recent message history for a channel"""
        if channel not in self.channels:
            return []
            
        return self.channels[channel].messages[-limit:]
        
    async def subscribe_to_channel(self, channel: ChannelType, websocket):
        """Subscribe a WebSocket to channel updates"""
        if channel not in self.websocket_connections:
            self.websocket_connections[channel] = []
        self.websocket_connections[channel].append(websocket)
        
    async def unsubscribe_from_channel(self, channel: ChannelType, websocket):
        """Unsubscribe a WebSocket from channel updates"""
        if channel in self.websocket_connections:
            if websocket in self.websocket_connections[channel]:
                self.websocket_connections[channel].remove(websocket)

    await websocket.accept()
    channel_type = ChannelType(channel)
    
    # Subscribe to channel
    conversation_engine = websocket.app.state.conversation_engine
    await conversation_engine.subscribe_to_channel(channel_type, websocket)
    
    try:
        while True:
            data = await websocket.receive_text()
            # Handle incoming messages from clients
            message_data = json.loads(data)
            message = Message(**message_data)
            await conversation_engine.add_message(message)
            
    except WebSocketDisconnect:
        await conversation_engine.unsubscribe_from_channel(channel_type, websocket)

import os

class GLM46Client:
    def __init__(self):
        self.api_key = os.getenv("GLM46_API_KEY")
        self.base_url = os.getenv("GLM46_BASE_URL", "https://api.z.ai/v1")
        self.session: Optional[aiohttp.ClientSession] = None
        
    async def initialize(self):
        self.session = aiohttp.ClientSession(
            headers={
                "Authorization": f"Bearer {self.api_key}",
                "Content-Type": "application/json"
            }
        )
        
    async def generate_code(self, prompt: str, context: Dict[str, Any]) -> str:
        """Generate code using GLM-4.6"""
        full_prompt = self._build_code_prompt(prompt, context)
        
        payload = {
            "model": "glm-4-6",
            "messages": [
                {
                    "role": "system",
                    "content": "You are an expert coding assistant. Generate clean, efficient, well-documented code. Always consider security and performance."
                },
                {
                    "role": "user", 
                    "content": full_prompt
                }
            ],
            "max_tokens": 4000,
            "temperature": 0.1,
            "top_p": 0.9
        }
        
        async with self.session.post(f"{self.base_url}/chat/completions", json=payload) as response:
            result = await response.json()
            return result["choices"][0]["message"]["content"]
            
    def _build_code_prompt(self, requirements: str, context: Dict[str, Any]) -> str:
        """Build a comprehensive prompt for code generation"""
        return f"""
        Generate code for the following requirements:
        
        REQUIREMENTS:
        {requirements}

        1. Complete, runnable code

        4. Error handling

import asyncio
from typing import Dict, List
import re
import ast
import aiohttp

from app.models.automation import *
from app.services.glm46_client import GLM46Client

class AutomationEngine:
    def __init__(self):
        self.glm46_client = GLM46Client()
        self.active_scripts: Dict[str, ScriptDeployment] = {}
        self.code_validator = CodeValidator()
        
    async def initialize(self):
        await self.glm46_client.initialize()
        
    async def request_automation(self, request: AutomationRequest) -> AutomationResponse:
        """Process an automation request"""
        try:
            # Generate code using GLM-4.6
            generated_code = await self.glm46_client.generate_code(
                request.requirements, 
                request.context
            )
            
            # Validate the generated code
            validation = await self.code_validator.validate_code(
                generated_code, 
                request.context
            )
            
            if not validation.is_safe:
                return AutomationResponse(
                    success=False,
                    errors=validation.issues
                )
                
            # Deploy the script
            script_id = await self.deploy_script(
                request.agent_id, 
                generated_code
            )
            
            return AutomationResponse(
                success=True,
                code=generated_code,
                warnings=validation.suggestions
            )
            
        except Exception as e:
            return AutomationResponse(
                success=False,
                errors=[f"Automation failed: {str(e)}"]
            )
            
    async def deploy_script(self, agent_id: str, code: str) -> str:
        """Deploy a script to an agent"""
        script_id = f"script_{agent_id}_{len(self.active_scripts)}"
        
        deployment = ScriptDeployment(
            script_id=script_id,
            agent_id=agent_id,
            code=code,
            deployed_at=datetime.now(),
            status="active"
        )
        
        self.active_scripts[script_id] = deployment
        return script_id

class CodeValidator:
    async def validate_code(self, code: str, context: Dict[str, Any]) -> CodeValidationResult:
        """Validate generated code for safety and effectiveness"""
        issues = []
        suggestions = []
        
        # Security checks
        dangerous_patterns = [
            r"eval\s*\(",
            r"exec\s*\(",
            r"__import__",
            r"os\.system",
            r"subprocess\.call",
            r"open\s*\(",
        ]
        
        for pattern in dangerous_patterns:
            if re.search(pattern, code):
                issues.append(f"Potentially dangerous pattern detected: {pattern}")
                
        # Basic syntax check for Python
        try:
            ast.parse(code)
        except SyntaxError as e:
            issues.append(f"Syntax error: {str(e)}")
            
        # Effectiveness suggestions
        if len(code.split('\n')) > 100:
            suggestions.append("Consider breaking code into smaller functions")
            
        if "TODO" in code or "FIXME" in code:
            suggestions.append("Code contains unfinished sections")
            
        return CodeValidationResult(
            is_safe=len(issues) == 0,
            is_effective=len(issues) < 3,  # Allow minor issues
            issues=issues,
            suggestions=suggestions
        )

import React, { useState, useEffect, useRef } from 'react';
import { useWebSocket } from '../hooks/useWebSocket';
import { Message, ChannelType } from '../types';

interface ConversationInterfaceProps {
  channel: ChannelType;
  agentId?: string;
}

export const ConversationInterface: React.FC<ConversationInterfaceProps> = ({ 
  channel, 
  agentId 
}) => {
  const [messages, setMessages] = useState<Message[]>([]);
  const [newMessage, setNewMessage] = useState('');
  const messagesEndRef = useRef<HTMLDivElement>(null);
  
  const { sendMessage, lastMessage } = useWebSocket(`ws://localhost:8000/api/v1/conversation/ws/${channel}`);
  
  useEffect(() => {
    if (lastMessage) {
      const message = JSON.parse(lastMessage.data) as Message;
      setMessages(prev => [...prev, message]);
    }
  }, [lastMessage]);
  
  useEffect(() => {
    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth' });
  }, [messages]);
  
  const handleSendMessage = () => {
    if (newMessage.trim() && agentId) {
      const message: Message = {
        id: Math.random().toString(36).substr(2, 9),
        sender_id: agentId,
        sender_type: 'agent',
        channel,
        content: newMessage,
        timestamp: new Date().toISOString()
      };
      
      sendMessage(JSON.stringify(message));
      setNewMessage('');
    }
  };
  
  return (
    <div className="flex flex-col h-full border rounded-lg">
      <div className="p-4 border-b">
        <h3 className="text-lg font-semibold">Channel: {channel}</h3>
      </div>
      
      <div className="flex-1 overflow-y-auto p-4 space-y-2">
        {messages.map((message) => (
          <div key={message.id} className="flex flex-col">
            <div className="flex items-center space-x-2">
              <span className="font-medium text-blue-600">{message.sender_id}</span>
              <span className="text-xs text-gray-500">
                {new Date(message.timestamp).toLocaleTimeString()}
              </span>
            </div>
            <p className="text-gray-800">{message.content}</p>
          </div>
        ))}
        <div ref={messagesEndRef} />
      </div>
      
      <div className="p-4 border-t">
        <div className="flex space-x-2">
          <input
            type="text"
            value={newMessage}
            onChange={(e) => setNewMessage(e.target.value)}
            onKeyPress={(e) => e.key === 'Enter' && handleSendMessage()}
            placeholder="Type a message..."
            className="flex-1 border rounded px-3 py-2"
          />
          <button
            onClick={handleSendMessage}
            className="bg-blue-500 text-white px-4 py-2 rounded hover:bg-blue-600"
          >
            Send
          </button>
        </div>
      </div>
    </div>
  );
};

from typing import Dict, List
from app.models.dm import *

class DMService:
    def __init__(self, agent_framework, conversation_engine):
        self.agent_framework = agent_framework
        self.conversation_engine = conversation_engine
        self.modification_history: List[BehaviorModification] = []
        
    async def modify_agent_behavior(self, modification: BehaviorModification) -> bool:
        """Apply behavior modification to an agent"""
        agent = self.agent_framework.agents.get(modification.agent_id)
        if not agent:
            return False
            
        try:
            if modification.modification_type == "personality_adjustment":
                for trait, value in modification.adjustments.items():
                    if hasattr(agent.personality, trait):
                        setattr(agent.personality, trait, value)
                        
            elif modification.modification_type == "goal_override":
                agent.active_goals = modification.adjustments.get("goals", [])
                
            elif modification.modification_type == "capability_grant":
                for capability, value in modification.adjustments.items():
                    if hasattr(agent.capabilities, capability):
                        setattr(agent.capabilities, capability, value)
                        
            self.modification_history.append(modification)
            
            # Notify the agent of changes
            await self.notify_agent_of_changes(agent, modification)
            
            return True
            
        except Exception as e:
            print(f"Behavior modification failed: {e}")
            return False
            
    async def create_dynamic_event(self, event_config: DynamicEventConfig):
        """Create a dynamic world event"""
        # Broadcast event to relevant channels
        event_message = f"WORLD EVENT: {event_config.name} - {event_config.description}"
        
        message = Message(
            id=f"event_{datetime.now().timestamp()}",
            sender_id="dm_system",
            sender_type="system", 
            channel=ChannelType.META,
            content=event_message,
            metadata={"event_config": event_config.model_dump()}
        )
        
        await self.conversation_engine.add_message(message)
        
        # Apply event effects to world state
        await self.apply_event_effects(event_config)
        
        # Schedule event expiration
        if event_config.duration > 0:
            asyncio.get_event_loop().call_later(
                event_config.duration,
                lambda: self.expire_event(event_config)
            )
