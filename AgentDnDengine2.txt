from typing import Dict, List, Optional, Set
from enum import Enum
import asyncio
from fastapi import WebSocket

class ControlMode(str, Enum):
    AGENT = "agent"
    HUMAN = "human"
    HYBRID = "hybrid"  # Human makes key decisions, agent handles routine

class CharacterControlService:
    def __init__(self):
        self.character_controls: Dict[str, ControlMode] = {}
        self.terminal_connections: Dict[str, List[WebSocket]] = {}
        self.character_states: Dict[str, Dict] = {}
        self.control_queues: Dict[str, asyncio.Queue] = {}
    
    async def take_character_control(self, character_id: str, user_id: str, mode: ControlMode) -> bool:
        """Allow a human to take control of a character"""
        if character_id in self.character_controls:
            # Character already controlled - could implement permission system
            current_mode = self.character_controls[character_id]
            if current_mode == ControlMode.HUMAN:
                return False  # Already controlled by someone else
        
        self.character_controls[character_id] = mode
        
        # Notify the agent to pause autonomous behavior
        await self.notify_agent_control_change(character_id, mode)
        
        # Log the control transfer
        await self.log_control_transfer(character_id, user_id, mode)
        
        return True
    
    async def release_character_control(self, character_id: str, user_id: str) -> bool:
        """Release character back to agent control"""
        if character_id not in self.character_controls:
            return False
        
        self.character_controls[character_id] = ControlMode.AGENT
        
        # Notify agent to resume autonomous behavior
        await self.notify_agent_control_change(character_id, ControlMode.AGENT)
        
        # Log release
        await self.log_control_release(character_id, user_id)
        
        return True
    
    async def process_character_input(self, character_id: str, input_text: str, source: str) -> Dict:
        """Process input for a character, routing based on control mode"""
        control_mode = self.character_controls.get(character_id, ControlMode.AGENT)
        
        if control_mode == ControlMode.HUMAN and source == "agent":
            # Agent trying to act while human in control - queue or ignore
            return {"status": "queued", "reason": "human_control"}
        
        elif control_mode == ControlMode.AGENT and source == "human":
            # Human giving command while agent in control - temporary override
            return await self.process_human_override(character_id, input_text)
        
        elif control_mode == ControlMode.HUMAN and source == "human":
            # Direct human control
            return await self.execute_human_command(character_id, input_text)
        
        else:  # AGENT + agent input
            return await self.process_agent_action(character_id, input_text)
    
    async def broadcast_to_terminal(self, character_id: str, message: str, message_type: str = "info"):
        """Broadcast message to all terminal connections for this character"""
        if character_id in self.terminal_connections:
            payload = {
                "type": "terminal_output",
                "line": f"[{message_type.upper()}] {message}",
                "timestamp": datetime.now().isoformat()
            }
            
            for websocket in self.terminal_connections[character_id]:
                try:
                    await websocket.send_json(payload)
                except:
                    # Remove broken connections
                    self.terminal_connections[character_id].remove(websocket)
    
    async def connect_terminal(self, character_id: str, websocket: WebSocket):
        """Connect a WebSocket to character's terminal"""
        await websocket.accept()
        
        if character_id not in self.terminal_connections:
            self.terminal_connections[character_id] = []
        
        self.terminal_connections[character_id].append(websocket)
        
        # Send welcome message and current state
        await self.broadcast_to_terminal(
            character_id,

from typing import Dict, List, Optional
import os
import json

class CoderWorkshopService:
    def __init__(self):
        self.script_storage = ScriptStorage()
        self.glm46_client = GLM46Client()
        self.script_executor = ScriptExecutor()
    
    async def request_coder_assistance(self, request: CoderRequest) -> Dict:
        """Request code generation from GLM-4.6 coder agent"""
        
        prompt = self.build_coder_prompt(request)
        
        try:
            generated_code = await self.glm46_client.generate_code(prompt, request.context)
            
            # Validate the generated code
            validation = await self.validate_generated_code(generated_code, request.context)
            
            if validation["is_safe"]:
                # Store the generated script
                script_id = await self.script_storage.store_script(
                    name=f"Coder-Generated-{datetime.now().strftime('%Y%m%d-%H%M%S')}",
                    code=generated_code,
                    language=request.context.get("preferred_language", "python"),
                    status="draft",
                    metadata={
                        "generated_by": "coder_agent",
                        "requirements": request.requirements,
                        "validation_results": validation
                    }
                )
                
                return {
                    "success": True,
                    "script_id": script_id,
                    "generated_code": generated_code,
                    "warnings": validation.get("warnings", [])
                }
            else:
                return {
                    "success": False,
                    "error": "Generated code failed safety validation",
                    "issues": validation.get("issues", [])
                }
                
        except Exception as e:
            return {
                "success": False,
                "error": f"Coder request failed: {str(e)}"
            }
    
    def build_coder_prompt(self, request: CoderRequest) -> str:
        """Build a comprehensive prompt for the coder agent"""
        return f"""
        Create a D&D 5e automation script with the following requirements:
        
        REQUIREMENTS:
        {request.requirements}
        
        CONTEXT:
        Character: {request.context.get('character_id', 'Unknown')}
        Game State: {json.dumps(request.context.get('game_state', {}), indent=2)}
        Existing Scripts: {len(request.context.get('current_scripts', []))}
        
        TECHNICAL CONSTRAINTS:
        - Language: {request.context.get('preferred_language', 'python')}
        - Must use the D&D 5e rule engine API
        - Include error handling
        - Consider character abilities and skills
        - Optimize for the current game situation
        
        Please provide:
        1. Complete, runnable code
        2. Brief documentation
        3. Usage examples
        4. Error handling for edge cases
        
        Ensure the code is safe, efficient, and follows D&D 5e rules.
        """
    
    async def test_script(self, script: AutomationScript, test_context: Dict) -> ScriptExecutionResult:
        """Test a script in a safe environment"""
        try:
            result = await self.script_executor.execute_safely(script.code, test_context)
            return {
                "success": True,
                "output": result.get("output", ""),
                "execution_time": result.get("execution_time", 0),
                "logs": result.get("logs", [])
            }
        except Exception as e:
            return {
                "success": False,
                "error": str(e),
                "output": "",
                "logs": []
            }
    
    async def deploy_script_to_character(self, script_id: str, character_id: str) -> bool:
        """Deploy a script to a character's automation system"""
        script = await self.script_storage.get_script(script_id)
        if not script:
            return False

        success = await character_service.add_automation(character_id, script)
        
        if success:
            await self.script_storage.update_script_status(script_id, "active")
        
        return success

from typing import Dict, List
import asyncio
from fastapi import FastAPI, WebSocket
import portastic

class PortalGatewayService:
    def __init__(self):
        self.character_ports: Dict[str, int] = {}
        self.dm_portals: Dict[str, int] = {}
        self.available_ports = set(range(9000, 10000))
        
    async def create_character_portal(self, character_id: str) -> int:
        """Create a dedicated portal for a character"""
        if character_id in self.character_ports:
            return self.character_ports[character_id]
        
        port = await self.find_available_port()
        if not port:
            raise Exception("No available ports for character portal")
        
        self.character_ports[character_id] = port
        await self.start_character_portal(character_id, port)
        
        return port
    
    async def create_dm_portal(self, campaign_id: str) -> int:
        """Create a DM portal for a campaign"""
        if campaign_id in self.dm_portals:
            return self.dm_portals[campaign_id]
        
        port = await self.find_available_port()
        if not port:
            raise Exception("No available ports for DM portal")
        
        self.dm_portals[campaign_id] = port
        await self.start_dm_portal(campaign_id, port)
        
        return port
    
    async def find_available_port(self) -> int:
        """Find an available port in the range"""
        for port in self.available_ports:
            if await self.is_port_available(port):
                self.available_ports.discard(port)
                return port
        return None
    
    async def is_port_available(self, port: int) -> bool:
        """Check if a port is available"""
        checker = portastic.PortChecker()
        return checker.is_available(port, '0.0.0.0')
