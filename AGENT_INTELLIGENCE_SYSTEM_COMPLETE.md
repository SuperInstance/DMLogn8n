## 🎯 **Usage Examples**

### **Starting an Agent Learning Session**

```python
from agent_intelligence_system.learning_orchestrator_v2 import AdvancedLearningOrchestrator

# Initialize orchestrator
orchestrator = AdvancedLearningOrchestrator()
await orchestrator.initialize()

# Start learning session for character
session_id = await orchestrator.start_agent_session(
    agent_id="character_123",
    context={
        "class": "rogue",
        "level": 5,
        "personality": {
            "openness": 0.8,
            "conscientiousness": 0.6,
            "extraversion": 0.4,
            "agreeableness": 0.7,
            "neuroticism": 0.3
        },
        "campaign_id": "shadow_rising"
    }
)

# Process gameplay experiences
experience = {
    "agent_id": "character_123",
    "type": "combat",
    "skills_used": ["stealth", "sneak_attack", "acrobatics"],
    "success": True,
    "difficulty": 0.7,
    "importance": 0.8,
    "outcome": {
        "damage_dealt": 24,
        "critical_hit": True,
        "enemies_defeated": 1
    },
    "context": {
        "environment": "dark_forest",
        "surprised_enemy": True,
        "strategic_advantage": True
    }
}

# Submit for processing
result = await orchestrator.process_experience(experience)
print(f"Experience {result['experience_id']} queued for processing")
```

### **Monitoring Agent Progress**

```python
# Get comprehensive learning status
status = await orchestrator.get_agent_learning_status("character_123")

print(f"Intelligence Score: {status['learning_state']['intelligence_score']:.1f}")
print(f"Total Experiences: {status['metrics']['total_experiences']}")
print(f"Learning Velocity: {status['learning_velocity']:.2f} IQ/hour")

# View skill improvements
for skill_name, skill_data in status['skill_status'].items():
    if skill_data.get('level', 1) > 1:
        print(f"Skill: {skill_name} - Level {skill_data['level']} ({skill_data['xp']}/{skill_data['xp_next']})")

# Check memory status
memory_status = status['memory_status']
print(f"Working Memory: {memory_status['working_memory_count']}/{memory_status['working_capacity']}")
print(f"Episodic Memory: {memory_status['episodic_memory_count']}/{memory_status['episodic_capacity']}")
print(f"Recent Consolidations: {len(memory_status['recent_consolidations'])}")
```

### **Quick Deployment**

```bash

cd /home/activeloguser/DMlogn8n/agent-intelligence-system/

# 2. Install Python dependencies
pip install -r requirements.txt

# 3. Install dashboard dependencies
cd learning-dashboard/
npm install
cd ..

# 4. Start databases with Docker

python scripts/migrate_databases.py
```
