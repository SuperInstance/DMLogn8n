class AgentLoraTrainer:

        lora_config = LoraConfig(
            task_type=TaskType.CAUSAL_LM,
            inference_mode=False,
            r=8,  # Rank
            lora_alpha=32,
            lora_dropout=0.1,
            target_modules=["q_proj", "v_proj"]  # Model-specific
        )
        
        # Load base model
        model = AutoModelForCausalLM.from_pretrained(self.base_model_name)
        tokenizer = AutoTokenizer.from_pretrained(self.base_model_name)
        
        # Apply LoRA
        lora_model = get_peft_model(model, lora_config)
        
        # Save agent-specific adapter
        adapter_path = f"./data/agents/{agent_id}/lora_adapters"
        lora_model.save_pretrained(adapter_path)
        tokenizer.save_pretrained(adapter_path)
        
        self.lora_configs[agent_id] = {
            "adapter_path": adapter_path,
            "training_examples": 0,
            "last_trained": None,
            "performance_metrics": {}
        }
        
        return adapter_path
    
    async def collect_training_data(self, agent_id: str, experience: Dict):
        """Collect experiences for fine-tuning"""
        if agent_id not in self.training_data:
            self.training_data[agent_id] = []
        
        training_example = self.format_experience_for_training(experience)
        self.training_data[agent_id].append(training_example)
        
        # Check if we have enough data for training
        if len(self.training_data[agent_id]) >= 100:  # Batch size
            await self.train_agent_lora(agent_id)
    
    def format_experience_for_training(self, experience: Dict) -> Dict:
        """Format game experience for model training"""
        return {
            "input": self.build_training_input(experience),
            "output": self.build_training_output(experience),
            "metadata": {
                "success_metric": experience.get("success", 0),
                "context": experience.get("context", {}),
                "timestamp": experience.get("timestamp")
            }
        }
    
    def build_training_input(self, experience: Dict) -> str:
        """Build training input from experience"""
        context = experience.get("context", {})
        return f"""
        As {context.get('character_name', 'Unknown')}, a {context.get('character_class', 'adventurer')} with these traits: {context.get('personality_traits', [])}
        
        Situation: {experience.get('situation', 'Unknown')}
        Available actions: {experience.get('available_actions', [])}
        Recent events: {experience.get('recent_events', [])}
        
        What would be the most strategic action or response?
        """

        """Build training output from successful experiences"""
        successful_actions = experience.get("successful_actions", [])
        if successful_actions:
            # Use the most successful action
            best_action = max(successful_actions, key=lambda x: x.get("success_score", 0))
            return best_action.get("action", "")
        return experience.get("chosen_action", "")

                outputs = model(**batch)
                loss = outputs.loss
                
                # Backward pass
                optimizer.zero_grad()
                loss.backward()
                optimizer.step()
                
                total_loss += loss.item()
            
            avg_loss = total_loss / len(formatted_data)

    def calculate_memory_capacity(self, level: int) -> Dict[str, int]:
        """Calculate memory capacity based on character level"""
        base_capacity = {
            "episodic": 1000,
            "semantic": 500, 
            "procedural": 300,
            "social": 200
        }
        
        # Scale capacity with level
        scale_factor = 1 + (level - 1) * 0.2  # 20% increase per level
        
        return {k: int(v * scale_factor) for k, v in base_capacity.items()}
    
    async def store_experience(self, agent_id: str, experience: Dict, memory_type: str = "episodic"):
        """Store an experience in the appropriate memory collection"""
        if agent_id not in self.memory_collections:
            await self.initialize_agent_memory(agent_id)
        
        collection = self.memory_collections[agent_id].get(memory_type)
        if not collection:
            raise ValueError(f"Unknown memory type: {memory_type}")
        
        # Check capacity and prune if necessary
        await self.manage_memory_capacity(agent_id, memory_type)
        
        # Create memory embedding
        embedding = self.create_memory_embedding(experience, memory_type)
        memory_id = f"{memory_type}_{datetime.now().timestamp()}"
        
        # Store memory
        collection.add(
            embeddings=[embedding],
            documents=[self.experience_to_document(experience)],
            metadatas=[{
                "timestamp": datetime.now().isoformat(),
                "importance": experience.get("importance", 0.5),
                "emotional_valence": experience.get("emotional_valence", 0),
                "success_score": experience.get("success_score", 0),
                "tags": experience.get("tags", [])
            }],
            ids=[memory_id]
        )
        
        return memory_id
    
    async def retrieve_relevant_memories(self, agent_id: str, query: str, 
                                       memory_types: List[str] = None,
                                       max_memories: int = 10) -> List[Dict]:
        """Retrieve relevant memories for current context"""
        if memory_types is None:
            memory_types = ["episodic", "semantic", "procedural", "social"]
        
        all_memories = []
        
        for memory_type in memory_types:
            if memory_type not in self.memory_collections.get(agent_id, {}):
                continue
                
            collection = self.memory_collections[agent_id][memory_type]
            query_embedding = self.create_query_embedding(query)
            
            results = collection.query(
                query_embeddings=[query_embedding],
                n_results=max_memories // len(memory_types)
            )
            
            for i, doc in enumerate(results['documents'][0]):
                memory = {
                    "content": doc,
                    "type": memory_type,
                    "metadata": results['metadatas'][0][i],
                    "similarity": results['distances'][0][i] if results['distances'] else 0,
                    "id": results['ids'][0][i]
                }
                all_memories.append(memory)
        
        # Sort by relevance and importance
        all_memories.sort(key=lambda x: (
            x["metadata"].get("importance", 0.5) * 0.7 + 
            (1 - x["similarity"]) * 0.3  # Higher similarity = lower distance
        ), reverse=True)
        
        return all_memories[:max_memories]
    
    async def manage_memory_capacity(self, agent_id: str, memory_type: str):
        """Manage memory capacity by pruning less important memories"""
        collection = self.memory_collections[agent_id][memory_type]
        capacity = collection.metadata.get("capacity", 1000)
        
        # Get current count
        current_count = collection.count()
        
        if current_count >= capacity * 0.9:  # Start pruning at 90% capacity
            # Get all memories sorted by importance and recency
            all_memories = collection.get(include=["metadatas", "documents", "embeddings"])
            
            # Calculate retention score for each memory
            retention_scores = []
            for i, metadata in enumerate(all_memories["metadatas"]):
                importance = metadata.get("importance", 0.5)
                timestamp = datetime.fromisoformat(metadata.get("timestamp", datetime.now().isoformat()))
                recency = (datetime.now() - timestamp).days
                
                # Score formula: importance * recency_decay
                recency_decay = max(0, 1 - (recency / 30))  # Decay over 30 days
                retention_score = importance * recency_decay
                
                retention_scores.append((i, retention_score))
            
            # Sort by retention score (lowest first)
            retention_scores.sort(key=lambda x: x[1])
            
            # Remove lowest scoring memories (20% over capacity)
            memories_to_remove = int(current_count - (capacity * 0.8))
            memories_to_remove = max(1, memories_to_remove)
            
            ids_to_remove = [all_memories["ids"][i] for i, _ in retention_scores[:memories_to_remove]]
            
            if ids_to_remove:
                collection.delete(ids=ids_to_remove)

                    memory_type=self.determine_memory_type(experience)
                )

        emotional_intensity = abs(experience.get("emotional_valence", 0))

        if experience.get("strategic_impact", False):

        return min(1.0, importance)
