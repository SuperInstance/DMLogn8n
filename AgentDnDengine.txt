```python
from pydantic import BaseModel, Field
from typing import List, Dict, Optional, Set
from enum import Enum

class AbilityScore(str, Enum):
    STRENGTH = "strength"
    DEXTERITY = "dexterity"
    CONSTITUTION = "constitution"
    INTELLIGENCE = "intelligence"
    WISDOM = "wisdom"
    CHARISMA = "charisma"

class Skill(str, Enum):
    ACROBATICS = "acrobatics"
    ANIMAL_HANDLING = "animal_handling"
    ARCANA = "arcana"
    ATHLETICS = "athletics"
    DECEPTION = "deception"
    HISTORY = "history"
    INSIGHT = "insight"
    INTIMIDATION = "intimidation"
    INVESTIGATION = "investigation"
    MEDICINE = "medicine"
    NATURE = "nature"
    PERCEPTION = "perception"
    PERFORMANCE = "performance"
    PERSUASION = "persuasion"
    RELIGION = "religion"
    SLEIGHT_OF_HAND = "sleight_of_hand"
    STEALTH = "stealth"
    SURVIVAL = "survival"

class DnDCharacter(BaseModel):
    # Basic Info
    name: str
    race: str
    character_class: str
    level: int = Field(ge=1, le=20, default=1)
    background: str
    
    # Ability Scores
    strength: int = Field(ge=1, le=30, default=10)
    dexterity: int = Field(ge=1, le=30, default=10)
    constitution: int = Field(ge=1, le=30, default=10)
    intelligence: int = Field(ge=1, le=30, default=10)
    wisdom: int = Field(ge=1, le=30, default=10)
    charisma: int = Field(ge=1, le=30, default=10)
    
    # Derived Stats
    proficiency_bonus: int = Field(default=2)
    armor_class: int = Field(default=10)
    hit_points: int
    current_hit_points: int
    speed: int = Field(default=30)
    
    # Skills & Proficiencies
    skill_proficiencies: Set[Skill] = Field(default_factory=set)
    saving_throw_proficiencies: Set[AbilityScore] = Field(default_factory=set)
    
    # Combat
    initiative: int = Field(default=0)
    attacks: List[Dict] = Field(default_factory=list)
    spells: List[Dict] = Field(default_factory=list)
    
    # Inventory
    equipment: List[str] = Field(default_factory=list)
    gold: int = Field(default=0)
    
    # Roleplaying
    personality_traits: List[str] = Field(default_factory=list)
    ideals: List[str] = Field(default_factory=list)
    bonds: List[str] = Field(default_factory=list)
    flaws: List[str] = Field(default_factory=list)
    
    @property
    def ability_modifiers(self) -> Dict[AbilityScore, int]:
        return {
            AbilityScore.STRENGTH: (self.strength - 10) // 2,
            AbilityScore.DEXTERITY: (self.dexterity - 10) // 2,
            AbilityScore.CONSTITUTION: (self.constitution - 10) // 2,
            AbilityScore.INTELLIGENCE: (self.intelligence - 10) // 2,
            AbilityScore.WISDOM: (self.wisdom - 10) // 2,
            AbilityScore.CHARISMA: (self.charisma - 10) // 2,
        }
    
    def get_skill_modifier(self, skill: Skill) -> int:
        """Calculate skill modifier including proficiency"""
        ability_map = {
            Skill.ACROBATICS: AbilityScore.DEXTERITY,
            Skill.ANIMAL_HANDLING: AbilityScore.WISDOM,
            Skill.ARCANA: AbilityScore.INTELLIGENCE,
            Skill.ATHLETICS: AbilityScore.STRENGTH,
            Skill.DECEPTION: AbilityScore.CHARISMA,
            Skill.HISTORY: AbilityScore.INTELLIGENCE,
            Skill.INSIGHT: AbilityScore.WISDOM,
            Skill.INTIMIDATION: AbilityScore.CHARISMA,
            Skill.INVESTIGATION: AbilityScore.INTELLIGENCE,
            Skill.MEDICINE: AbilityScore.WISDOM,
            Skill.NATURE: AbilityScore.INTELLIGENCE,
            Skill.PERCEPTION: AbilityScore.WISDOM,
            Skill.PERFORMANCE: AbilityScore.CHARISMA,
            Skill.PERSUASION: AbilityScore.CHARISMA,
            Skill.RELIGION: AbilityScore.INTELLIGENCE,
            Skill.SLEIGHT_OF_HAND: AbilityScore.DEXTERITY,
            Skill.STEALTH: AbilityScore.DEXTERITY,
            Skill.SURVIVAL: AbilityScore.WISDOM,
        }
        
        ability_mod = self.ability_modifiers[ability_map[skill]]
        if skill in self.skill_proficiencies:
            return ability_mod + self.proficiency_bonus
        return ability_mod
```

```python
import random
from typing import Dict, List, Optional, Tuple
from app.models.dnd_character import *

class DnDRuleEngine:
    def __init__(self):
        self.conditions = self.initialize_conditions()
        
    def ability_check(self, character: DnDCharacter, ability: AbilityScore, 
                     advantage: bool = False, disadvantage: bool = False) -> Dict:
        """Perform an ability check following D&D 5e rules"""
        roll = self.roll_d20(advantage, disadvantage)
        modifier = character.ability_modifiers[ability]
        total = roll + modifier
        
        return {
            "roll": roll,
            "modifier": modifier,
            "total": total,
            "success": total >= 15,  # DC 15 for medium difficulty
            "critical_success": roll == 20,
            "critical_failure": roll == 1
        }
    
    def skill_check(self, character: DnDCharacter, skill: Skill,
                   advantage: bool = False, disadvantage: bool = False) -> Dict:
        """Perform a skill check"""
        roll = self.roll_d20(advantage, disadvantage)
        modifier = character.get_skill_modifier(skill)
        total = roll + modifier
        
        return {
            "skill": skill,
            "roll": roll,
            "modifier": modifier,
            "total": total,
            "success": total >= 15,
            "critical_success": roll == 20,
            "critical_failure": roll == 1
        }
    
    def attack_roll(self, attacker: DnDCharacter, target: DnDCharacter,
                   advantage: bool = False, disadvantage: bool = False) -> Dict:
        """Make an attack roll against target AC"""
        roll = self.roll_d20(advantage, disadvantage)
        attack_bonus = attacker.ability_modifiers[AbilityScore.STRENGTH] + attacker.proficiency_bonus
        total = roll + attack_bonus
        hits = total >= target.armor_class
        critical = roll == 20
        
        damage = 0
        if hits:
            # Simple damage calculation - in real implementation, use weapon stats
            damage_dice = "1d8"  # Default weapon damage
            damage = self.roll_dice(damage_dice) + attacker.ability_modifiers[AbilityScore.STRENGTH]
            if critical:
                damage += self.roll_dice(damage_dice)  # Double dice on crit
        
        return {
            "attack_roll": roll,
            "attack_bonus": attack_bonus,
            "total": total,
            "target_ac": target.armor_class,
            "hits": hits,
            "critical": critical,
            "damage": damage
        }
    
    def saving_throw(self, character: DnDCharacter, ability: AbilityScore,
                    dc: int = 15, advantage: bool = False, disadvantage: bool = False) -> Dict:
        """Perform a saving throw"""
        roll = self.roll_d20(advantage, disadvantage)
        modifier = character.ability_modifiers[ability]
        if ability in character.saving_throw_proficiencies:
            modifier += character.proficiency_bonus
        total = roll + modifier
        success = total >= dc
        
        return {
            "saving_throw": ability,
            "roll": roll,
            "modifier": modifier,
            "total": total,
            "dc": dc,
            "success": success
        }
    
    def roll_d20(self, advantage: bool = False, disadvantage: bool = False) -> int:
        """Roll a d20 with optional advantage/disadvantage"""
        if advantage and disadvantage:
            return random.randint(1, 20)  # Normal roll if both
        
        roll1 = random.randint(1, 20)
        roll2 = random.randint(1, 20)
        
        if advantage:
            return max(roll1, roll2)
        elif disadvantage:
            return min(roll1, roll2)
        else:
            return roll1
    
    def roll_dice(self, dice_string: str) -> int:
        """Roll dice in format like '2d6', '1d8', etc."""
        if 'd' not in dice_string:
            return int(dice_string)
        
        num_dice, dice_sides = map(int, dice_string.split('d'))
        return sum(random.randint(1, dice_sides) for _ in range(num_dice))
    
    def calculate_proficiency_bonus(self, level: int) -> int:
        """Calculate proficiency bonus based on level"""
        return 2 + (level - 1) // 4
    
    def initialize_conditions(self) -> Dict:
        """Initialize D&D 5e conditions"""
        return {
            "blinded": {"disadvantage_attack": True, "advantage_attacked": True},
            "charmed": {"cannot_attack_charmer": True},
            "frightened": {"disadvantage_ability_checks": True},
            "grappled": {"speed_zero": True},
            "incapacitated": {"cannot_take_actions": True},
            "invisible": {"advantage_attack": True, "disadvantage_attacked": True},
            "poisoned": {"disadvantage_attack_checks": True},
            "prone": {"disadvantage_attack": True, "advantage_melee_attacked": True},
            "restrained": {"disadvantage_attack": True, "advantage_attacked": True},
            "stunned": {"cannot_take_actions": True, "advantage_attacked": True},
            "unconscious": {"cannot_take_actions": True, "advantage_attacked": True}
        }
```

```python
from typing import Dict, List, Optional
from app.models.dnd_character import *
from app.services.dnd_engine import DnDRuleEngine

class DnDConversationEngine:
    def __init__(self):
        self.rule_engine = DnDRuleEngine()
        self.active_encounters: Dict[str, DnDEncounter] = {}
        
    async def process_dnd_action(self, agent_id: str, action: str, 
                               targets: List[str], context: Dict) -> Dict:
        """Process a D&D action from conversation"""
        action_lower = action.lower()
        
        if any(word in action_lower for word in ["attack", "strike", "hit", "smash"]):
            return await self.process_attack_action(agent_id, targets, context)
        elif any(word in action_lower for word in ["cast", "spell", "magic"]):
            return await self.process_spell_action(agent_id, action, targets, context)
        elif any(word in action_lower for word in ["persuade", "convince", "deceive", "intimidate"]):
            return await self.process_social_action(agent_id, action, targets, context)
        elif any(word in action_lower for word in ["search", "investigate", "perceive"]):
            return await self.process_investigation_action(agent_id, action, context)
        else:
            return await self.process_general_action(agent_id, action, context)
    
    async def process_attack_action(self, agent_id: str, targets: List[str], context: Dict) -> Dict:
        """Process an attack action"""
        attacker = await self.get_character(agent_id)
        if not targets:
            return {"success": False, "error": "No target specified"}
        
        target_id = targets[0]
        target = await self.get_character(target_id)
        
        attack_result = self.rule_engine.attack_roll(attacker, target)
        
        if attack_result["hits"]:
            target.current_hit_points -= attack_result["damage"]
            await self.update_character(target)
            
            return {
                "type": "attack",
                "success": True,
                "attacker": attacker.name,
                "target": target.name,
                "attack_roll": attack_result["attack_roll"],
                "total_attack": attack_result["total"],
                "damage": attack_result["damage"],
                "target_hp": target.current_hit_points,
                "critical": attack_result["critical"],
                "narrative": self.generate_attack_narrative(attacker, target, attack_result)
            }
        else:
            return {
                "type": "attack",
                "success": False,
                "attacker": attacker.name,
                "target": target.name,
                "attack_roll": attack_result["attack_roll"],
                "total_attack": attack_result["total"],
                "target_ac": target.armor_class,
                "narrative": f"{attacker.name} attacks {target.name} but misses!"
            }
    
    async def process_social_action(self, agent_id: str, action: str, 
                                  targets: List[str], context: Dict) -> Dict:
        """Process persuasion, deception, or intimidation"""
        actor = await self.get_character(agent_id)
        
        if "persuade" in action.lower() or "convince" in action.lower():
            skill = Skill.PERSUASION
        elif "deceive" in action.lower() or "lie" in action.lower():
            skill = Skill.DECEPTION
        elif "intimidate" in action.lower():
            skill = Skill.INTIMIDATION
        else:
            skill = Skill.PERSUASION
        
        skill_check = self.rule_engine.skill_check(actor, skill)
        
        # Determine DC based on target attitudes and context
        dc = self.calculate_social_dc(context, skill)
        
        success = skill_check["total"] >= dc
        
        return {
            "type": "social",
            "skill": skill,
            "actor": actor.name,
            "targets": targets,
            "skill_check": skill_check,
            "dc": dc,
            "success": success,
            "narrative": self.generate_social_narrative(actor, skill, skill_check, success, targets)
        }
    
    def generate_attack_narrative(self, attacker: DnDCharacter, target: DnDCharacter, 
                                result: Dict) -> str:
        """Generate narrative description of attack"""
        if result["critical"]:
            return (f"{attacker.name} lands a devastating critical hit on {target.name}, "
                   f"dealing {result['damage']} damage! {target.name} staggers from the blow.")
        elif result["hits"]:
            return (f"{attacker.name} strikes {target.name} true, dealing {result['damage']} damage. "
                   f"{target.name} has {result['target_hp']} HP remaining.")
        else:
            return f"{attacker.name} attacks but {target.name} deftly avoids the blow!"
    
    def generate_social_narrative(self, actor: DnDCharacter, skill: Skill,
                                check: Dict, success: bool, targets: List[str]) -> str:
        """Generate narrative for social actions"""
        target_str = ", ".join(targets) if targets else "the group"
        
        skill_descriptions = {
            Skill.PERSUASION: "makes a compelling argument to",
            Skill.DECEPTION: "attempts to deceive",
            Skill.INTIMIDATION: "tries to intimidate"
        }
        
        if success:
            if check["critical_success"]:
                return (f"{actor.name} {skill_descriptions[skill]} {target_str} with masterful skill! "
                       f"They are completely convinced.")
            else:
                return (f"{actor.name} {skill_descriptions[skill]} {target_str} successfully. "
                       f"Their words have the desired effect.")
        else:
            if check["critical_failure"]:
                return (f"{actor.name} {skill_descriptions[skill]} {target_str} but fails spectacularly! "
                       f"The attempt backfires horribly.")
            else:
                return (f"{actor.name} {skill_descriptions[skill]} {target_str} but fails to convince them. "
                       f"The attempt falls flat.")
```

```python
        prompt = f"""
        As a D&D 5e Dungeon Master, adjudicate this action:
        
        Action: {action}
        Context: {context}
        
        Consider:
        1. What ability check or saving throw is required?
        2. What should the DC be?

        """Generate a D&D quest using AI"""
        quest = await self.quest_generator.generate_quest(party_composition, setting)
        return quest
    
    async def manage_initiative(self, characters: List[DnDCharacter]) -> List[Dict]:
        """Manage combat initiative order"""
        initiative_order = []
        
        for character in characters:
            initiative_roll = self.rule_engine.ability_check(
                character, AbilityScore.DEXTERITY
            )
            initiative_order.append({
                "character": character,
                "initiative_roll": initiative_roll["total"],
                "roll_details": initiative_roll
            })
        
        # Sort by initiative (highest first)
        initiative_order.sort(key=lambda x: x["initiative_roll"], reverse=True)
        
        return initiative_order

            "medium": ["orc", "hobgoblin", "wolf"],
            "hard": ["ogre", "troll", "young dragon"],
            "deadly": ["adult dragon", "lich", "beholder"]
        }
        
        selected_monsters = []
        for monster_name in monster_pool[difficulty]:
            selected_monsters.append({
                "name": monster_name,
                "stats": self.get_monster_stats(monster_name),
                "count": random.randint(1, 4)
            })
        
        return selected_monsters
```

```typescript
  const [gameLog, setGameLog] = useState<GameEvent[]>([]);

  const handleDnDAction = async (action: string, targets: string[]) => {

    const response = await fetch('/api/v1/dnd/action', {

      headers: { 'Content-Type': 'application/json' },

        action,
        targets,
        context: {} // Add game context
      })
    });

    // Add to game log
    setGameLog(prev => [...prev, {
      type: 'dnd_action',
      timestamp: new Date(),
      data: result,
      narrative: result.narrative
    }]);

    if (result.character_updates) {
      setActiveCharacter(prev => prev ? { ...prev, ...result.character_updates } : null);
    }
  };

    <div className="h-screen flex bg-medieval bg-cover">
      {/* Left Panel - Character Sheet */}

        {activeCharacter && (

        )}
      </div>

      {/* Center Panel - Game Log */}
      <div className="flex-1 p-4">
        <DnDConversationLog 
          events={gameLog}
          onSendMessage={handleDnDAction}
        />
      </div>
```

```python
from typing import Dict, List
import random

class DnDStrategyEngine:
    def __init__(self):
        self.personality_templates = self.load_personality_templates()
        self.dialogue_patterns = self.load_dialogue_patterns()
    
    async def generate_character_strategy(self, character: DnDCharacter, 
                                       situation: Dict) -> Dict:
        """Generate strategic approach based on character personality and situation"""
        
        # Analyze situation
        threat_level = self.assess_threat_level(situation)
        available_resources = self.assess_resources(character, situation)
        
        # Determine optimal approach based on character personality
        if character.character_class == "rogue":
            strategy = await self.generate_rogue_strategy(character, situation)
        elif character.character_class == "wizard":
            strategy = await self.generate_wizard_strategy(character, situation)
        elif character.character_class == "fighter":
            strategy = await self.generate_fighter_strategy(character, situation)
        else:
            strategy = await self.generate_general_strategy(character, situation)
        
        return strategy
    
    async def generate_rogue_strategy(self, character: DnDCharacter, 
                                    situation: Dict) -> Dict:
        """Generate strategy for rogue characters"""
        return {
            "primary_approach": "stealth_and_subterfuge",
            "preferred_skills": [Skill.STEALTH, Skill.SLEIGHT_OF_HAND, Skill.DECEPTION],
            "combat_tactics": ["sneak_attack", "flank", "disengage"],
            "social_approach": "manipulative",
            "dialogue_style": "clever_and_sarcastic",
            "risk_tolerance": "calculated"
        }
    
    async def generate_dialogue_response(self, character: DnDCharacter, 
                                       incoming_message: str, context: Dict) -> str:
        """Generate character-appropriate dialogue response"""
        
        prompt = f"""
        As {character.name}, a {character.race} {character.character_class} with these personality traits:
        Traits: {', '.join(character.personality_traits)}
        Ideals: {', '.join(character.ideals)}
        Bonds: {', '.join(character.bonds)}
        Flaws: {', '.join(character.flaws)}
        
        Current situation: {context.get('situation', 'general interaction')}
        
        Another character says: "{incoming_message}"
        
        Respond in character, being clever and strategic. Consider:
        - Your character's goals and personality
        - The game mechanics and your abilities
        - How to gain advantage in the situation
        - Potential alliances or rivalries
        
        Keep response under 3 sentences and focus on clever wordplay or strategic positioning.
        """
        
        response = await self.glm46_client.generate_dialogue(prompt)
        return response
    
    def assess_threat_level(self, situation: Dict) -> str:
        """Assess how dangerous the current situation is"""
        enemies = situation.get("enemies", [])
        party_health = situation.get("party_health", 1.0)
        
        if not enemies:
            return "low"
        elif party_health < 0.3:
            return "critical"
        elif party_health < 0.6:
            return "high"
        else:
            return "medium"
```
